VP::Bind
  size 8 align 8
  virtual Command command(std::string_view, std::string_view)
  virtual File file(std::string_view)
  virtual std::string folder() const
  virtual bool sets(std::string_view)
  virtual std::string name() const
  virtual std::string view_name() const
  virtual const Engine & engine() const
  virtual std::uint32_t value_offset(std::string_view, std::string_view)
  virtual std::uint32_t readback_index(std::string_view, const Element &)
  virtual std::uint32_t upload_index(std::string_view, const Element &, std::uint32_t)
  virtual std::uint32_t texture_index(std::string_view, std::string_view)
  virtual std::string_view param_text(std::string_view)
  virtual void param_read_as(std::string_view, std::string_view, bool)
  virtual void * output_object(std::string_view, const Kind &)
  virtual const void * input_object(std::string_view, const Kind &)
  virtual const View & view() const

  virtual std::span<const NodePort> ports() const
